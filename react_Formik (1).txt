import React, { useState } from  'react'
import {useFormik}  from 'formik'

function  FillingForm()
{

   const  initialValues={
        name:'ravi',
        email:'r33@gmail.com',
        pet:'Dog'
        }
        const onSubmit=values =>{
            console.log("form Data",values)
            }

            const  validate =  values =>{
                let errors={}
                if(!values.ename){
                errors.name='Required'
                }
                if(!values.email){
                errors.email='Required'
                }
                
                if(!values.pet){
                errors.pet='Required'
                }
                return  errors;
                }

    const formik=useFormik(
{
initialValues,
onSubmit,
validate
}
)
return(
<div>
<form onSubmit={formik.handleSubmit}>
<label>Name</label>
<input type='text'  name='name'  onChange={formik.handleChange} value={formik.values.name}/>
{formik.errors.name? <div>{formik.errors.name}</div>:null}
<label>Email</label>
<input type='text'  name='email'  onChange={formik.handleChange} value={formik.values.email}/>
{formik.errors.email? <div>{formik.errors.email}</div>:null}
<label>Pet</label>
<input type='text'  name='pet'  onChange={formik.handleChange} value={formik.values.pet}/>
{formik.errors.pet? <div>{formik.errors.pet}</div>:null}

<button type="submit">Submit</button>




</form>
</div>

)


}


   

        export  default  FillingForm;
