let arr1 = [1, 2, 3, 4, 5];
console.log("Array: " + arr1.join(", "));
let arr2 = ["Hello", "World", "TypeScript"];
console.log("Array: " + arr2.join(", "));
arr1.push(6);
console.log("Updated Array: " + arr1.join(", "));
arr2.pop();
console.log("Updated Array: " + arr2.join(", "));
for (let i = 0; i < arr1.length; i++) {
    console.log("Element at index " + i + ": " + arr1[i]);
}
arr1.forEach((num, index) => {
    console.log("Element at index " + index + ": " + num);
});
let mappedArr = arr1.map(num => num * 2);
console.log("Mapped Array (doubled): " + mappedArr.join(", "));
let filteredArr = arr1.filter(num => num % 2 === 0);
console.log("Filtered Array (even numbers): " + filteredArr.join(", "));
let reducedValue = arr1.reduce((acc, num) => acc + num, 0);
console.log("Reduced Value (sum): " + reducedValue);

interface hospital_login {
    username: string;
    password: string;
}
let patient_info = {
    patient_name: "Jane Smith",
    patient_email: "jane@  example.com",
    patient_phone: "987-654-3210",
    patient_address: "456 Elm St, Othertown, USA"
};

const hospitalLogin = (user: hospital_login): void => {
    if (user.username === "admin" && user.password === "123") {
        console.log("Login successful!");
        console.log("Patient Information:");
        console.log("Name: " + patient_info.patient_name);
        console.log("Email: " + patient_info.patient_email);
        console.log("Phone: " + patient_info.patient_phone);
        console.log("Address: " + patient_info.patient_address);
    } else {
        console.log("Login failed. Invalid username or password.");
        return;
    }       
}
