<!DOCTYPE html>
<html>
<body>

<h2>Employee Table</h2>

<input id="name" placeholder="Enter Name">

  <th>Name</th>
  <th>Action</th>

</table>

<script>
function addRow() {
    let name = document.getElementById("name").value;

    let table = document.getElementById("empTable");
    let row = table.insertRow();

    let cell1 = row.insertCell(0);
    let cell2 = row.insertCell(1);

    cell1.innerHTML = name;

    let btn = document.createElement("button");
    btn.innerHTML = "Delete";

    btn.onclick = function() {
        table.deleteRow(row.rowIndex);
    };

    cell2.appendChild(btn);
}
</script>

</body>
</html>






Real-Time Example: Live Search Filter

👉 Used in:

E-commerce search
Tables filtering



<!DOCTYPE html>
<html>
<body>

<input type="text" id="search" onkeyup="filter()" placeholder="Search">

<ul id="list">
  <li>Apple</li>
  <li>Banana</li>
  <li>Orange</li>
</ul>

<script>
function filter() {
    let input = document.getElementById("search").value.toLowerCase();
    let items = document.querySelectorAll("#list li");

    items.forEach(function(item) {
        if(item.innerHTML.toLowerCase().includes(input)) {
            item.style.display = "";
        } else {
            item.style.display = "none";
        }
    });
}
</script>

</body>
</html>

function loadData() {
    fetch("https://jsonplaceholder.typicode.com/users")
    .then(response => response.json())
    .then(data => {
        let list = document.getElementById("users");

        data.forEach(user => {
            let li = document.createElement("li");
            li.innerHTML = user.name;
            list.appendChild(li);
        });
    });
}
</script>

    localStorage.removeItem("loggedInUser");

    }

🔷 2. Real-Time Example: User List (CRUD)

👉 Simulates adding/removing users (like admin panel)

<!DOCTYPE html>
<html>
<body>

<h2>User Management</h2>

<input type="text" id="username" placeholder="Enter name">
<button onclick="addUser()">Add</button>
<button onclick="showUsers()">Show</button>

<ul id="list"></ul>

<script>
let users = [];

function addUser() {
    let name = document.getElementById("username").value;
    users.push(name);
    alert("User Added");
}

function showUsers() {
    let list = document.getElementById("list");
    list.innerHTML = "";

    users.forEach((user, index) => {
        let li = document.createElement("li");
        li.innerText = user + " ";
        
        let btn = document.createElement("button");
        btn.innerText = "Delete";
        btn.onclick = () => deleteUser(index);

        li.appendChild(btn);
        list.appendChild(li);
    });
}

function deleteUser(index) {
    users.splice(index, 1);
    showUsers();
}
</script>

</body>
</html>

    let value = document.getElementById("search").value.toLowerCase();

    let filtered = products.filter(p => p.toLowerCase().includes(value));

    let list = document.getElementById("result");
    list.innerHTML = "";

2. How to Setup Web SQL (Step-by-Step)
Step 1: Create HTML File

Create a file:
websql-setup.html

Step 2: Add Web SQL Code
<!DOCTYPE html>
<html>
<body>

<h2>Web SQL Setup Example</h2>

<button onclick="initDB()">Init DB</button>
<button onclick="insertData()">Insert</button>
<button onclick="viewData()">View</button>

<ul id="result"></ul>

<script>
let db;

// Step 1: Open/Create Database
function initDB() {
    db = openDatabase('myDB', '1.0', 'Test Database', 2 * 1024 * 1024);

    db.transaction(function(tx) {
        tx.executeSql(
            'CREATE TABLE IF NOT EXISTS users (id INTEGER PRIMARY KEY, name)'
        );
    });

    alert("Database Initialized");
}

// Step 2: Insert Data
function insertData() {
    db.transaction(function(tx) {
        tx.executeSql(
            'INSERT INTO users (name) VALUES (?)',
            ['Babu']
        );
    });

    alert("Data Inserted");
}

// Step 3: Fetch Data
function viewData() {
    db.transaction(function(tx) {
        tx.executeSql('SELECT * FROM users', [], function(tx, results) {

            let output = document.getElementById("result");
            output.innerHTML = "";

            for (let i = 0; i < results.rows.length; i++) {
                let row = results.rows.item(i);
                let li = document.createElement("li");
                li.innerText = row.id + " - " + row.name;
                output.appendChild(li);
            }
        });
    });
}
</script>

</body>
</html>

👉 Browser doesn’t support Web SQL

❌ Data not saving

👉 You didn’t initialize DB first
✔️ Click "Init DB" before insert
