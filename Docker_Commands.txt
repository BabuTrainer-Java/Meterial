package com.example;

public class HelloDocker {

    public static void main(String[] args) {

        System.out.println("Hello from Docker!");
    }
}




FROM eclipse-temurin:17-jdk

WORKDIR /app

COPY . .

RUN javac -d . src/com/example/HelloDocker.java

CMD ["java", "com.example.HelloDocker"]


docker build -t java-package-demo .


docker run java-package-demo





package com.example.demo;

import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RestController;

@RestController
public class HelloController {

    @GetMapping("/")
    public String home() {

        return "Hello from Spring Boot Docker!";
    }

    @GetMapping("/hello")
    public String hello() {

        return "Spring Boot application is running inside Docker!";
    }
}



mvn clean package

target/
└── spring-docker-demo-1.0.0.jar



FROM eclipse-temurin:17-jre

WORKDIR /app

COPY target/spring-docker-demo-1.0.0.jar app.jar

EXPOSE 8080

ENTRYPOINT ["java", "-jar", "app.jar"]



docker build -t spring-docker-demo .

docker run -d --name spring-container -p 8080:8080 spring-docker-demo
